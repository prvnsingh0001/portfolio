SELECT plant_name,
       SUM(achieved_qty) AS total_qty,
       RANK() OVER (
         PARTITION BY plant_name
         ORDER BY SUM(achieved_qty) DESC
       ) AS plant_rank
FROM production_log
WHERE txn_date >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY plant_name
ORDER BY total_qty DESC;
